# Pinagem DE1 (Cyclone II EP2C20F484C7) para bombom01
# Use: Assignments > Import Assignments (ou cole no .qsf com o Quartus fechado)

# clock e botões
set_location_assignment PIN_L1  -to clk
set_location_assignment PIN_R22 -to reset_n
set_location_assignment PIN_R21 -to moeda_inserida_n
set_location_assignment PIN_T22 -to confirma_selecao_n
set_location_assignment PIN_T21 -to desiste_n

# chaves: produto SW0..SW2
set_location_assignment PIN_L22 -to sel_produto[0]
set_location_assignment PIN_L21 -to sel_produto[1]
set_location_assignment PIN_M22 -to sel_produto[2]

# chaves: estoque SW3..SW7
set_location_assignment PIN_V12 -to estoque[0]
set_location_assignment PIN_W12 -to estoque[1]
set_location_assignment PIN_U12 -to estoque[2]
set_location_assignment PIN_U11 -to estoque[3]
set_location_assignment PIN_M2  -to estoque[4]

# chaves: moeda SW8..SW9
set_location_assignment PIN_M1  -to moeda_valor[0]
set_location_assignment PIN_L2  -to moeda_valor[1]

# LEDs verdes
set_location_assignment PIN_U22 -to libera_salgado
set_location_assignment PIN_U21 -to libera_troco
set_location_assignment PIN_V22 -to libera_todas_as_moedas
set_location_assignment PIN_W21 -to moedas_aceitas[0]
set_location_assignment PIN_Y22 -to moedas_aceitas[1]
set_location_assignment PIN_Y21 -to moedas_aceitas[2]

# LEDs vermelhos
set_location_assignment PIN_R20 -to aviso_sem_estoque
set_location_assignment PIN_R19 -to moeda_invalida

# HEX0: saldo, primeiro decimal
set_location_assignment PIN_J2  -to display_decimo[0]
set_location_assignment PIN_J1  -to display_decimo[1]
set_location_assignment PIN_H2  -to display_decimo[2]
set_location_assignment PIN_H1  -to display_decimo[3]
set_location_assignment PIN_F2  -to display_decimo[4]
set_location_assignment PIN_F1  -to display_decimo[5]
set_location_assignment PIN_E2  -to display_decimo[6]

# HEX1: saldo, reais
set_location_assignment PIN_E1  -to display_reais[0]
set_location_assignment PIN_H6  -to display_reais[1]
set_location_assignment PIN_H5  -to display_reais[2]
set_location_assignment PIN_H4  -to display_reais[3]
set_location_assignment PIN_G3  -to display_reais[4]
set_location_assignment PIN_D2  -to display_reais[5]
set_location_assignment PIN_D1  -to display_reais[6]

# HEX2: troco, primeiro decimal
set_location_assignment PIN_G5  -to troco_decimo[0]
set_location_assignment PIN_G6  -to troco_decimo[1]
set_location_assignment PIN_C2  -to troco_decimo[2]
set_location_assignment PIN_C1  -to troco_decimo[3]
set_location_assignment PIN_E3  -to troco_decimo[4]
set_location_assignment PIN_E4  -to troco_decimo[5]
set_location_assignment PIN_D3  -to troco_decimo[6]

# HEX3: troco, reais
set_location_assignment PIN_F4  -to troco_reais[0]
set_location_assignment PIN_D5  -to troco_reais[1]
set_location_assignment PIN_D6  -to troco_reais[2]
set_location_assignment PIN_J4  -to troco_reais[3]
set_location_assignment PIN_L8  -to troco_reais[4]
set_location_assignment PIN_F3  -to troco_reais[5]
set_location_assignment PIN_D4  -to troco_reais[6]
